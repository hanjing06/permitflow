<objective>
Take Plan 03-02's `clusters.json` and Plan 03-03's `conflict-graph.json`, produce the two timelines (`naive.json` + `optimized.json`) and the durable headline metrics (`metrics.json`).

This is the synthesis step — the optimizer's actual *output* that Phase 4 visualizes. The "naive vs optimized" toggle in the UI is just two reads of these files.

Greedy scheduler algorithm:
- For each cluster record in clusters.json, the optimized timeline emits ONE event per record (collapsed) at the anchor's window (piggyback/anchor-only) or at the merged_window (merge) or at the candidate's original window (singleton).
- Then walk the optimized events in start_date order. For each event, check `conflict-graph.json` for active conflicts (`delta_seconds >= threshold`) against any currently-active event. If conflict, defer the new event to the earliest slot after the conflicting event ends.

Metrics:
- `permits_considered` = total candidate + anchor rows (matches clusters.json invariant)
- `excavations_avoided` = sum of clusters.json `excavations_avoided` field
- `lane_days_saved` = sum of (sum of member durations) - (cluster's merged duration) across piggyback + merge records
- `estimated_cost_avoidance_usd` = `lane_days_saved * PER_LANE_DAY_COST` where the constant is a documented hackathon placeholder (D-26/D-27 — real number lands in Phase 6)

Purpose: complete the four-of-five JSON contract (clusters + conflict + naive + optimized + metrics) so Phase 4 has everything it needs.
Output: `backend/scheduler.py`, `data/naive.json`, `data/optimized.json`, `data/metrics.json`, expanded SCHEMAS.md.

```python
"""
Greedy interval scheduler. Reads clusters.json + conflict-graph.json,
emits naive.json / optimized.json / metrics.json.
"""

import os
import json
from datetime import date, timedelta
from collections import defaultdict
import pandas as pd

DATA_DIR = os.path.join(os.path.dirname(__file__), "..", "data")
PER_LANE_DAY_COST = float(os.getenv("PERMITFLOW_PER_LANE_DAY_COST", "15000"))
CONFLICT_THRESHOLD_S = float(os.getenv("PERMITFLOW_CONFLICT_THRESHOLD_S", "30"))
DEFAULT_DURATION_DAYS = int(os.getenv("PERMITFLOW_DEFAULT_DURATION_DAYS", "7"))

def _load_inputs():
    with open(os.path.join(DATA_DIR, "clusters.json")) as f:
        clusters = json.load(f)
    cg_path = os.path.join(DATA_DIR, "conflict-graph.json")
    if os.path.exists(cg_path):
        with open(cg_path) as f:
            cg = json.load(f)
    else:
        cg = {"edges": [], "nodes": []}
    return clusters, cg

def _parse_date(s):
    if not s or s == "None" or pd.isna(s):
        return None
    return date.fromisoformat(s[:10])

def _ensure_end(start_d, end_d):
    if end_d is None and start_d is not None:
        return start_d + timedelta(days=DEFAULT_DURATION_DAYS)
    return end_d

def _lane_days(start_d, end_d):
    if start_d is None or end_d is None:
        return 0
    return max(1, (end_d - start_d).days + 1)

def build_naive_timeline(clusters):
    """One event per permit. No coordination. Each anchor + each member becomes its own event."""
    events = []
    seen = set()
    for rec in clusters:
        anchor = rec.get("anchor")
        if anchor and anchor["permit_id"] not in seen:
            seen.add(f"{anchor['source']}-{anchor['permit_id']}")
            s = _parse_date(anchor["start_date"])
            e = _ensure_end(s, _parse_date(anchor["end_date"]))
            events.append({
                "event_id": f"naive-{anchor['source']}-{anchor['permit_id']}",
                "tag": "naive",
                "permit_ids": [f"{anchor['source']}-{anchor['permit_id']}"],
                "anchor_id": None,
                "street_label": anchor["street_label"],
                "start_date": s.isoformat() if s else None,
                "end_date": e.isoformat() if e else None,
                "lane_days": _lane_days(s, e),
                "deferred_from": None,
                "deferred_reason": None,
            })
        for m in rec["members"]:
            key = f"{m['source']}-{m['permit_id']}"
            if key in seen: continue
            seen.add(key)
            s = _parse_date(m["start_date"])
            e = _ensure_end(s, _parse_date(m["end_date"]))
            events.append({
                "event_id": f"naive-{key}",
                "tag": "naive",
                "permit_ids": [key],
                "anchor_id": None,
                "street_label": m["street_label"],
                "start_date": s.isoformat() if s else None,
                "end_date": e.isoformat() if e else None,
                "lane_days": _lane_days(s, e),
                "deferred_from": None,
                "deferred_reason": None,
            })
    events.sort(key=lambda x: x["start_date"] or "9999")
    return events

def build_optimized_timeline(clusters, conflict_graph):
    """Collapsed per-cluster events, then conflict-deferred."""
    events = []
    for rec in clusters:
        permit_ids = []
        if rec.get("anchor"):
            permit_ids.append(f"{rec['anchor']['source']}-{rec['anchor']['permit_id']}")
        permit_ids.extend(f"{m['source']}-{m['permit_id']}" for m in rec["members"])

        if rec["tag"] == "piggyback":
            # shift into anchor's window
            s = _parse_date(rec["anchor"]["start_date"])
            e = _ensure_end(s, _parse_date(rec["anchor"]["end_date"]))
        elif rec["tag"] == "anchor-only":
            s = _parse_date(rec["anchor"]["start_date"])
            e = _ensure_end(s, _parse_date(rec["anchor"]["end_date"]))
        elif rec["tag"] == "merge":
            # merged window covers all members
            starts = [_parse_date(m["start_date"]) for m in rec["members"] if m["start_date"]]
            ends = [_ensure_end(_parse_date(m["start_date"]), _parse_date(m["end_date"]))
                    for m in rec["members"] if m["start_date"]]
            s = min(starts) if starts else None
            e = max(ends) if ends else None
        else:  # singleton
            m = rec["members"][0]
            s = _parse_date(m["start_date"])
            e = _ensure_end(s, _parse_date(m["end_date"]))

        events.append({
            "event_id": f"opt-{rec['cluster_id']}",
            "tag": rec["tag"],
            "permit_ids": permit_ids,
            "anchor_id": f"{rec['anchor']['source']}-{rec['anchor']['permit_id']}" if rec.get("anchor") else None,
            "street_label": (rec["anchor"]["street_label"] if rec.get("anchor")
                            else rec["members"][0]["street_label"]),
            "start_date": s.isoformat() if s else None,
            "end_date": e.isoformat() if e else None,
            "lane_days": _lane_days(s, e),
            "deferred_from": None,
            "deferred_reason": None,
        })

    # Conflict-defer pass: sort by start, walk; if any active event has a conflict edge
    # with delta_seconds >= threshold, push start to (active.end + 1d).
    edges = {(e["permit_a"], e["permit_b"]): e["delta_seconds"] for e in conflict_graph["edges"]
             if e["delta_seconds"] >= CONFLICT_THRESHOLD_S}
    # symmetric lookup
    for (a, b), d in list(edges.items()):
        edges[(b, a)] = d

    events.sort(key=lambda x: x["start_date"] or "9999")
    active = []  # list of (end_date, event)
    for ev in events:
        if not ev["start_date"]:
            continue
        ev_s = date.fromisoformat(ev["start_date"])
        ev_e = date.fromisoformat(ev["end_date"]) if ev["end_date"] else ev_s + timedelta(days=DEFAULT_DURATION_DAYS)

        # prune active
        active = [(e_end, e_obj) for (e_end, e_obj) in active if e_end >= ev_s]

        conflict_with = None
        for (a_end, a_ev) in active:
            for pa in ev["permit_ids"]:
                for pb in a_ev["permit_ids"]:
                    if (pa, pb) in edges:
                        conflict_with = a_ev
                        new_start = a_end + timedelta(days=1)
                        break
                if conflict_with: break
            if conflict_with: break

        if conflict_with:
            duration = (ev_e - ev_s).days
            ev["deferred_from"] = ev["start_date"]
            ev["deferred_reason"] = f"conflict:{conflict_with['event_id']}"
            ev["start_date"] = new_start.isoformat()
            ev["end_date"] = (new_start + timedelta(days=duration)).isoformat()
            ev_e = new_start + timedelta(days=duration)

        active.append((ev_e, ev))

    events.sort(key=lambda x: x["start_date"] or "9999")
    return events

def build_metrics(clusters, naive, optimized):
    permits_considered = sum(len(ev["permit_ids"]) for ev in naive)   # one permit per naive event
    excavations_avoided = sum(r["excavations_avoided"] for r in clusters)
    lane_days_naive = sum(ev["lane_days"] for ev in naive)
    # lane days for optimized: count merged windows once
    lane_days_opt = sum(ev["lane_days"] for ev in optimized)
    lane_days_saved = max(0, lane_days_naive - lane_days_opt)
    estimated_cost_avoidance = int(lane_days_saved * PER_LANE_DAY_COST)
    return {
        "permits_considered": int(permits_considered),
        "excavations_avoided": int(excavations_avoided),
        "lane_days_saved": int(lane_days_saved),
        "lane_days_naive": int(lane_days_naive),
        "lane_days_optimized": int(lane_days_opt),
        "estimated_cost_avoidance_usd": estimated_cost_avoidance,
        "per_lane_day_cost_usd": int(PER_LANE_DAY_COST),
        "per_lane_day_cost_source": "placeholder — Toronto Congestion Management Plan figure lands in Phase 6 (D-26)",
        "by_tag": {tag: sum(1 for r in clusters if r["tag"] == tag)
                   for tag in ("piggyback","merge","singleton","anchor-only")},
    }

def run():
    clusters, cg = _load_inputs()
    naive = build_naive_timeline(clusters)
    optimized = build_optimized_timeline(clusters, cg)
    metrics = build_metrics(clusters, naive, optimized)

    with open(os.path.join(DATA_DIR, "naive.json"), "w") as f:
        json.dump(naive, f, indent=2)
    with open(os.path.join(DATA_DIR, "optimized.json"), "w") as f:
        json.dump(optimized, f, indent=2)
    with open(os.path.join(DATA_DIR, "metrics.json"), "w") as f:
        json.dump(metrics, f, indent=2)

    print(json.dumps({
        "naive_events": len(naive),
        "optimized_events": len(optimized),
        "metrics": metrics,
    }, indent=2))

if __name__ == "__main__":
    run()
```
