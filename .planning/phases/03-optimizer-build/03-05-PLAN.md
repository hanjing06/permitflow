import os
import json
from typing import Optional, List

from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel

from llm import build_cluster_prompt, ask_local_llm

import data_prep
import valhalla_sim

DATA_DIR = os.path.join(os.path.dirname(__file__), "..", "data")

app = FastAPI(title="PermitFlow")
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],

    allow_headers=["*"],

def _read_json(name):
    path = os.path.join(DATA_DIR, name)
    if not os.path.exists(path):
        raise HTTPException(status_code=503, detail=f"{name} not yet generated — run Phase 3 pipeline (data_prep → trench_share → conflict_graph → scheduler)")
    with open(path) as f:
        return json.load(f)

    }
    return {
        "ok": True,
        "valhalla_alive": valhalla_sim.is_alive(),

    }

@app.get("/metrics")

class WhatIfBody(BaseModel):
    permit_id: Optional[str] = None
    polygon: Optional[List[List[float]]] = None  # [[lon,lat], ...]


@app.post("/whatif")
def whatif(body: WhatIfBody):
    """
    Real Valhalla closure simulation.
    Accepts either permit_id (looked up in clusters/anchors/candidates) OR explicit polygon.
    """
    if not body.permit_id and not body.polygon:
        raise HTTPException(status_code=400, detail="provide permit_id or polygon")

    if not valhalla_sim.is_alive():
        raise HTTPException(status_code=503, detail="Valhalla unreachable at :5000 — start the routing stack")

    geom_wkt = None
    if body.permit_id:
        # Look the permit up in anchors + candidates parquets to find its geometry
        import pandas as pd
        anchors = pd.read_parquet(os.path.join(DATA_DIR, "anchors.parquet"))
        candidates = pd.read_parquet(os.path.join(DATA_DIR, "candidates.parquet"))
        src, pid = body.permit_id.split("-", 1)
        for df in (anchors, candidates):
            hit = df[(df["source"] == src) & (df["permit_id"] == pid)]
            if not hit.empty and pd.notna(hit.iloc[0]["geom_wkt"]):
                geom_wkt = hit.iloc[0]["geom_wkt"]
                break
        if geom_wkt is None:
            raise HTTPException(status_code=404, detail=f"permit {body.permit_id} not found or has no geometry")
        result = valhalla_sim.simulate_closure(geom_wkt)
    else:
        # Wrap explicit polygon in the Valhalla-friendly nested list
        polys = [body.polygon]
        from valhalla_sim import _route, OD_PAIRS, baseline_seconds
        base = baseline_seconds()
        closed_total = 0.0
        unreachable = 0
        for o, d in OD_PAIRS:
            t = _route(o, d, exclude_polygons=polys)
            if t is None:
                closed_total += 3600.0
                unreachable += 1
            else:
                closed_total += t
        result = {
            "delta_seconds": round(closed_total - base, 1),
            "baseline_seconds": round(base, 1),
            "closed_seconds": round(closed_total, 1),
            "unreachable_pairs": unreachable,
        }

    # Tack on affected_permits from conflict-graph if available
    affected = []
    cg_path = os.path.join(DATA_DIR, "conflict-graph.json")
    if body.permit_id and os.path.exists(cg_path):
        with open(cg_path) as f:
            cg = json.load(f)
        for edge in cg.get("edges", []):
            if edge["permit_a"] == body.permit_id:
                affected.append({"permit_id": edge["permit_b"], "edge_delta_seconds": edge["delta_seconds"]})
            elif edge["permit_b"] == body.permit_id:
                affected.append({"permit_id": edge["permit_a"], "edge_delta_seconds": edge["delta_seconds"]})

    delta_min = result["delta_seconds"] / 60.0 if result["delta_seconds"] != float("inf") else "unreachable"
    result["affected_permits"] = affected
    result["narrative_hint"] = (
        f"Closing this segment adds ~{delta_min:.1f}min across hero-block O-D pairs"
        if isinstance(delta_min, float)
        else "Closure makes part of the hero block unreachable"
    )
    return result


# === Legacy endpoints (kept for backwards compat — current App.jsx uses these) ===

@app.get("/permits")
def get_permits():
    """
    Legacy. Returns the old open_toronto_permits.csv-derived view if present,
    else returns a flat list derived from anchors + candidates parquets so the
    existing map renderer keeps working.
    """
    legacy_csv = os.path.join(DATA_DIR, "open_toronto_permits.csv")
    if os.path.exists(legacy_csv):
        df = load_permits()
        df = cluster_permits(df)
        cols = ["permit_id", "street_name", "work_type", "status", "start_year", "lat", "lon", "cluster_id"]
        return df[cols].head(500).to_dict(orient="records")

    import pandas as pd
    anchors = pd.read_parquet(os.path.join(DATA_DIR, "anchors.parquet"))
    candidates = pd.read_parquet(os.path.join(DATA_DIR, "candidates.parquet"))
    combined = pd.concat([anchors, candidates], ignore_index=True)
    combined["permit_id"] = combined.apply(lambda r: f"{r['source']}-{r['permit_id']}", axis=1)
    combined["street_name"] = combined["street_label"]
    combined["status"] = combined["status"].fillna("Unknown")
    combined["start_year"] = combined["start_date"].apply(lambda d: getattr(d, "year", None))
    combined["cluster_id"] = -1
    cols = ["permit_id", "street_name", "work_type", "status", "start_year", "lat", "lon", "cluster_id"]
    return combined[cols].fillna("").head(500).to_dict(orient="records")


@app.get("/recommendations")
def get_recommendations():
    """Legacy. Reshapes clusters.json into the old recommendations array."""
    clusters_data = _read_json("clusters.json")
    out = []
    for r in clusters_data:
        if r["tag"] in ("singleton", "anchor-only"):
            continue
        out.append({
            "cluster_id": r["cluster_id"],
            "tag": r["tag"],
            "permit_count": r["permits_in_record"],
            "road_openings_saved": r["excavations_avoided"],
            "locations": [m["street_label"] for m in r["members"]][:8],
            "projects": list({m["work_type"] for m in r["members"]})[:5],
            "statuses": list({m.get("status", "Unknown") for m in r["members"]}),
            "years": sorted({m["start_date"][:4] for m in r["members"] if m.get("start_date")}),
            "action": "Coordinate this trench-share opportunity." if r["tag"] == "piggyback" else "Merge these adjacent permits into one closure.",
            "reason": "Anchor program window — utility work can piggyback for zero extra disruption." if r["tag"] == "piggyback" else "Permits cluster spatially+temporally; one closure replaces multiple.",
            "priority": "High" if r["excavations_avoided"] >= 3 else "Medium" if r["excavations_avoided"] >= 1 else "Low",
            "estimated_savings": r["excavations_avoided"] * 15000,
        })
    return {"recommendations": out}


@app.get("/explain")
def explain(cluster_id: str):
    """Legacy. cluster_id is now a string like 'anchor-road_resurfacing-7' or 'merge-3'."""
    clusters_data = _read_json("clusters.json")
    rec = next((r for r in clusters_data if r["cluster_id"] == cluster_id), None)
    if rec is None:
        return {"error": "Cluster not found"}
    # Adapt to the existing prompt shape
    proxy = {
        "cluster_id": rec["cluster_id"],
        "permit_count": rec["permits_in_record"],
        "road_openings_saved": rec["excavations_avoided"],
        "estimated_savings": rec["excavations_avoided"] * 15000,
        "priority": "High" if rec["excavations_avoided"] >= 3 else "Medium",
        "locations": [m["street_label"] for m in rec["members"]][:8],
        "projects": list({m["work_type"] for m in rec["members"]})[:5],
        "statuses": list({m.get("status","Unknown") for m in rec["members"]}),
    }
    prompt = build_cluster_prompt(proxy)
    explanation = ask_local_llm(prompt)
    return {"cluster_id": cluster_id, "source": "ollama", "cluster": rec, "explanation": explanation}
