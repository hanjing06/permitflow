`trip.summary.time` is seconds. `length` is km. For closed route vs baseline,
`delta_seconds = closed.time - baseline.time`. If `exclude_polygons` makes the
destination unreachable, Valhalla returns HTTP 400 with `error_code` — treat that
as `delta_seconds = +Inf` (severe conflict), record reason.

```python
"""
Valhalla closure simulator — POST /route with exclude_polygons.

Used by:
  - backend/conflict_graph.py     (precompute pass)
  - backend/main.py /whatif        (live endpoint, Plan 03-05)

Returns delta_seconds between baseline route and closed-route across
configured hero-block O-D pairs. Reachability failures count as +Inf.
"""

import os
import json
import math
import requests
from shapely import wkt
from shapely.geometry import LineString, MultiLineString

VALHALLA_URL = os.getenv("VALHALLA_URL", "http://localhost:5000/route")
VALHALLA_TIMEOUT = float(os.getenv("VALHALLA_TIMEOUT", "10"))
BUFFER_DEG = float(os.getenv("VALHALLA_BUFFER_DEG", "0.00015"))  # ~15m

# Hero-block O-D pairs (loaded from hero-block.json at import time)
HERO_BBOX_PATH = os.path.join(
    os.path.dirname(__file__), "..",
    ".planning/phases/01-parallel-kickoff/hero-block.json",
)

def _load_od_pairs():
    with open(HERO_BBOX_PATH) as f:
        hb = json.load(f)
    b = hb["bbox"]
    c = hb["center"]
    return [
        ((b["south"], b["west"]), (b["north"], b["east"])),
        ((b["south"], b["east"]), (b["north"], b["west"])),
        ((c["lat"], b["west"]),  (c["lat"], b["east"])),
    ]

OD_PAIRS = _load_od_pairs()

def _route(o, d, exclude_polygons=None):
    body = {
        "locations": [
            {"lat": o[0], "lon": o[1]},
            {"lat": d[0], "lon": d[1]},
        ],
        "costing": "auto",
        "directions_options": {"units": "kilometers"},
    }
    if exclude_polygons:
        body["exclude_polygons"] = exclude_polygons
    r = requests.post(VALHALLA_URL, json=body, timeout=VALHALLA_TIMEOUT)
    if r.status_code != 200:
        # Unreachable / blocked → caller treats as +Inf
        return None
    return r.json()["trip"]["summary"]["time"]

def _geom_to_polygon(geom_wkt):
    g = wkt.loads(geom_wkt)
    if isinstance(g, (LineString, MultiLineString)):
        poly = g.buffer(BUFFER_DEG)
    else:
        poly = g.buffer(BUFFER_DEG)
    # exclude_polygons wants list-of-rings-of-[lon,lat]
    if poly.geom_type == "Polygon":
        return [[list(c) for c in poly.exterior.coords]]
    elif poly.geom_type == "MultiPolygon":
        return [[list(c) for c in p.exterior.coords] for p in poly.geoms]
    return None

def baseline_seconds():
    """Sum of baseline route times across all OD pairs. Cached."""
    if hasattr(baseline_seconds, "_cache"):
        return baseline_seconds._cache
    total = 0.0
    for o, d in OD_PAIRS:
        t = _route(o, d)
        if t is not None:
            total += t
    baseline_seconds._cache = total
    return total

def simulate_closure(geom_wkt):
    """
    Returns dict {delta_seconds, baseline_seconds, closed_seconds, unreachable_pairs}.
    delta_seconds = closed - baseline. Unreachable pairs contribute INF (capped at 3600).
    """
    polys = _geom_to_polygon(geom_wkt)
    if not polys:
        return {"delta_seconds": 0, "baseline_seconds": baseline_seconds(),
                "closed_seconds": baseline_seconds(), "unreachable_pairs": 0,
                "reason": "no_polygon"}

    baseline = baseline_seconds()
    closed_total = 0.0
    unreachable = 0
    INF_PENALTY = 3600.0   # 1h penalty per unreachable pair
    for o, d in OD_PAIRS:
        t = _route(o, d, exclude_polygons=polys)
        if t is None:
            closed_total += INF_PENALTY
            unreachable += 1
        else:
            closed_total += t
    return {
        "delta_seconds": round(closed_total - baseline, 1),
        "baseline_seconds": round(baseline, 1),
        "closed_seconds": round(closed_total, 1),
        "unreachable_pairs": unreachable,
    }

def is_alive():
    """Quick health check used by conflict_graph.py to decide whether to attempt the precompute."""
    try:
        body = {"locations": [{"lat": 43.6657, "lon": -79.3642},
                              {"lat": 43.6680, "lon": -79.3580}],
                "costing": "auto"}
        r = requests.post(VALHALLA_URL, json=body, timeout=2)
        return r.status_code == 200
    except Exception:
        return False

if __name__ == "__main__":
    import sys
    print("Valhalla alive:", is_alive())
    if is_alive():
        print("Baseline seconds:", baseline_seconds())
```

```python
"""
Pairwise conflict adjacency for hero-block permits.

Reads data/anchors.parquet + data/candidates.parquet (Plan 03-01).
For each permit with geometry that falls inside the hero bbox:
  - Call valhalla_sim.simulate_closure(geom_wkt) once → record "solo" delta.
For each pair of such permits:
  - Build a combined polygon (union of the two) → simulate.
  - conflict_delta = combined_delta - solo_a_delta - solo_b_delta.
  - Positive conflict_delta = compounding congestion (real conflict).
  - Negative or zero = closures don't interact materially.

Writes data/conflict-graph.json. Designed to be re-runnable;
will take ~30min for ~50 permits (D-24 expected duration).
"""

import os
import json
import time
import pandas as pd
from shapely import wkt
from shapely.ops import unary_union

from valhalla_sim import simulate_closure, is_alive, _geom_to_polygon, _route, OD_PAIRS, baseline_seconds

DATA_DIR = os.path.join(os.path.dirname(__file__), "..", "data")

CONFLICT_THRESHOLD_S = float(os.getenv("PERMITFLOW_CONFLICT_THRESHOLD_S", "30"))
# Cap pair count for hackathon safety (the CONTEXT expects ~50 permits = ~1225 pairs;
# if dataset is much larger, sample down)
MAX_PERMITS = int(os.getenv("PERMITFLOW_MAX_PERMITS", "60"))

def _combined_polygon(geom_a_wkt, geom_b_wkt):
    a = wkt.loads(geom_a_wkt).buffer(0.00015)
    b = wkt.loads(geom_b_wkt).buffer(0.00015)
    u = unary_union([a, b])
    if u.geom_type == "Polygon":
        return [[list(c) for c in u.exterior.coords]]
    return [[list(c) for c in p.exterior.coords] for p in u.geoms]

def _combined_delta(polys):
    base = baseline_seconds()
    closed = 0.0
    for o, d in OD_PAIRS:
        t = _route(o, d, exclude_polygons=polys)
        if t is None:
            closed += 3600.0
        else:
            closed += t
    return round(closed - base, 1)

def build():
    if not is_alive():
        stub = {"nodes": [], "edges": [], "reason": "valhalla_unreachable",
                "baseline_seconds": None}
        out = os.path.join(DATA_DIR, "conflict-graph.json")
        with open(out, "w") as f:
            json.dump(stub, f, indent=2)
        print(f"Valhalla unreachable. Wrote stub → {out}")
        return stub

    anchors = pd.read_parquet(os.path.join(DATA_DIR, "anchors.parquet"))
    candidates = pd.read_parquet(os.path.join(DATA_DIR, "candidates.parquet"))

    geom_rows = pd.concat([
        anchors[anchors["in_hero_bbox"] & anchors["geom_wkt"].notna()],
        candidates[candidates["in_hero_bbox"] & candidates["geom_wkt"].notna()],
    ], ignore_index=True)

    if len(geom_rows) > MAX_PERMITS:
        geom_rows = geom_rows.head(MAX_PERMITS)

    print(f"Building conflict graph over {len(geom_rows)} permits...")

    # Solo deltas (cache so we don't recompute per pair)
    solo = {}
    for _, row in geom_rows.iterrows():
        key = f"{row['source']}-{row['permit_id']}"
        r = simulate_closure(row["geom_wkt"])
        solo[key] = r["delta_seconds"]
        print(f"  solo {key}: {r['delta_seconds']}s")

    # Pairwise
    rows = geom_rows.to_dict(orient="records")
    edges = []
    total_pairs = len(rows) * (len(rows) - 1) // 2
    done = 0
    t0 = time.time()
    for i in range(len(rows)):
        for j in range(i + 1, len(rows)):
            a = rows[i]; b = rows[j]
            ka = f"{a['source']}-{a['permit_id']}"
            kb = f"{b['source']}-{b['permit_id']}"
            polys = _combined_polygon(a["geom_wkt"], b["geom_wkt"])
            combined_d = _combined_delta(polys)
            conflict = round(combined_d - solo[ka] - solo[kb], 1)
            if conflict >= CONFLICT_THRESHOLD_S:
                edges.append({
                    "permit_a": ka,
                    "permit_b": kb,
                    "delta_seconds": conflict,
                    "combined_seconds": combined_d,
                    "solo_a_seconds": solo[ka],
                    "solo_b_seconds": solo[kb],
                    "shared_streets": [],   # left empty for hackathon; Phase 4 displays edges anyway
                })
            done += 1
            if done % 25 == 0:
                elapsed = time.time() - t0
                print(f"  {done}/{total_pairs} pairs done ({elapsed:.0f}s elapsed)")

    graph = {
        "baseline_seconds": baseline_seconds(),
        "threshold_seconds": CONFLICT_THRESHOLD_S,
        "nodes": [{"permit_id": k, "solo_delta_seconds": v} for k, v in solo.items()],
        "edges": edges,
    }
    out = os.path.join(DATA_DIR, "conflict-graph.json")
    with open(out, "w") as f:
        json.dump(graph, f, indent=2)
    print(f"Wrote {len(edges)} edges → {out}")
    return graph

if __name__ == "__main__":
    build()
```
