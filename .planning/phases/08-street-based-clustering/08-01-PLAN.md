<objective>
Replace the radius-based DBSCAN leftover-clustering pass in `backend/optimizer.py:cluster_leftover_candidates` with street-aware grouping (Layers 1+2 from CONTEXT — Layer 3 explicitly deferred). The optimizer's "merged cluster" recommendations should map to physical reality: permits cluster when they share the same opened road (same GEO_ID segment, or same normalized street name within a temporal window), not because they happen to be geometrically close on parallel streets.

Purpose: the radius-DBSCAN pass produces dishonest cluster counts (false-positive parallel-street merges, false-negative same-street splits). The D-79 trench-sharing thesis is *physically* about shared opened road, not euclidean proximity. This phase fixes the implementation to match the thesis.

Output: a new `backend/street_norm.py` helper module, a rewritten `cluster_leftover_candidates`, additive `geo_id` + `normalized_street` permit fields, a new additive `match_type` field on every recommendation in `clusters.json`, regenerated artifacts on disk, an additive `## Phase 8 additions` section in SCHEMAS.md, and an optional one-line `match: same_segment|same_street` subtitle in the cluster popup. Phase 4's TriStatCounter and ChatPanel continue to read the same metrics.json field names with no removals.

```python
STREET_TYPE_CANONICAL: dict[str, str]
# Maps abbreviations to canonical full form, e.g. "AVE" -> "AVENUE", "ST" -> "STREET",
# "RD" -> "ROAD", "BLVD" -> "BOULEVARD", "CRES" -> "CRESCENT", "DR" -> "DRIVE",
# "LN" -> "LANE", "PL" -> "PLACE", "CT" -> "COURT", "PKWY" -> "PARKWAY",
# "TR" -> "TRAIL", "TER" -> "TERRACE", "GDNS" -> "GARDENS", "SQ" -> "SQUARE",
# "CIR" -> "CIRCLE", "HTS" -> "HEIGHTS", "GRV" -> "GROVE", "WAY" -> "WAY",
# "PROM" -> "PROMENADE".
# Both directions (canonicalize "AVENUE" too) — see normalize_street below.

DIRECTION_PREFIXES: set[str] = {"E", "W", "N", "S", "NE", "NW", "SE", "SW"}

def normalize_street(raw: str | None) -> dict:
    """
    Pure helper. Input: raw street text from any source.
      - Anchor LOCATION (after clean_text): "DANFORTH AVE | From: PAPE AVE | To: BROADVIEW AVE"
      - utility_cuts DISPLAY_DESC: "154 ROBINA AVE (Between GLENHURST AVE AND EARLSDALE AVE)"
      - building_permits composed: "E DANFORTH AVE"
      - Empty / None / "Unknown location" / NaN.

    Returns:
      {
        "normalized": str,   # lowercased canonical key for grouping ("danforth avenue").
                             # Empty string for unparseable input (callers MUST treat empty as
                             # "no group" — singleton).
        "direction": str,    # "E", "W", "N", "S", "NE"... or "" if absent.
        "display": str,      # Cleaned human-readable form preserving casing
                             # ("DANFORTH AVE" not "danforth avenue") — for popups.
      }

    Algorithm:
      1. If raw is None / NaN / empty / "Unknown location" → return {"normalized": "",
         "direction": "", "display": ""}.
      2. Take only the FIRST segment of `raw.split(" | ")` — drops "From: X | To: Y" tails.
      3. Drop everything from the first "(" onward — drops "(Between A AND B)" tails.
      4. Strip leading house number: if the first token is all digits, drop it.
      5. Tokenize on whitespace. If the first token is in DIRECTION_PREFIXES, pop it into
         `direction`. Otherwise direction is "".
      6. If the last token is in STREET_TYPE_CANONICAL keys, replace with the canonical form.
         (Also: if the last token is already a canonical full form like "AVENUE", leave it.)
      7. Reassemble:
           display = " ".join(tokens).strip().upper()
           normalized = display.lower()
      8. Return.

    The empty-normalized case is a soft signal — `cluster_leftover_candidates` must treat
    permits with normalized=="" as singletons (no group). NEVER cluster all unparseable
    permits into one giant "unknown" cluster.
    """
```

    STREET_CLUSTER_WINDOW_DAYS = int(os.getenv("PERMITFLOW_STREET_WINDOW_DAYS", "60"))

    def cluster_leftover_candidates(permits, used_candidate_ids):
        candidates = [
            p for p in permits
            if p["role"] == "candidate" and p["permit_id"] not in used_candidate_ids
        ]
        if not candidates:
            return []

        # --- Layer 2: GEO_ID exact match (preferred, gold-standard segment ids) ---
        geo_groups = defaultdict(list)
        no_geo = []
        for p in candidates:
            gid = p.get("geo_id")
            if gid:
                geo_groups[gid].append(p)
            else:
                no_geo.append(p)

        clusters = []  # list of (match_type, list[permit])
        for gid, group in geo_groups.items():
            if len(group) >= 2:
                # Sub-bucket by date proximity even within a GEO_ID — two utility cuts on
                # the same segment 18 months apart are not the same trench event.
                for sub in _temporal_subclusters(group, STREET_CLUSTER_WINDOW_DAYS):
                    if len(sub) >= 2:
                        clusters.append(("same_segment", sub))
                # Singletons within a GEO_ID fall through to Layer 1 along with no_geo
                singletons = [p for sub in _temporal_subclusters(group, STREET_CLUSTER_WINDOW_DAYS) for p in sub if len(sub) < 2]
                no_geo.extend(singletons)
            else:
                no_geo.extend(group)

        # --- Layer 1: Same-street + temporal sub-clustering ---
        street_groups = defaultdict(list)
        for p in no_geo:
            key = p.get("normalized_street", "")
            if not key:
                continue  # never group permits with empty normalized_street
            street_groups[key].append(p)

        for key, group in street_groups.items():
            if len(group) < 2:
                continue
            for sub in _temporal_subclusters(group, STREET_CLUSTER_WINDOW_DAYS):
                if len(sub) >= 2:
                    clusters.append(("same_street", sub))

        # --- Convert to recommendation dicts (preserve clusters.json contract) ---
        recommendations = []
        for label, (match_type, group) in enumerate(clusters):
            start = min(p["start_date"] for p in group)
            end = max(p["end_date"] for p in group)
            saved = sum(p["lane_days"] for p in group) - max(p["lane_days"] for p in group)
            recommendations.append({
                "type": "merge",
                "cluster_id": f"merge:{label}",
                "match_type": match_type,
                "member_permit_ids": [p["permit_id"] for p in group],
                "merged_window": {
                    "start": serialize_date(start),
                    "end": serialize_date(end),
                },
                "savings_lane_days": int(saved),
                "lane_days_saved": int(saved),
                "excavations_avoided": len(group) - 1,
                "permit_count": len(group),
                "road_openings_saved": len(group) - 1,
                "estimated_savings": int(saved * LANE_DAY_COST),
                "priority": "High" if saved >= 20 else "Medium",
                "locations": sorted({p["street_name"] for p in group})[:8],
                "projects": sorted({p["work_type"] for p in group})[:5],
                "statuses": sorted({p["status"] for p in group}),
                "action": (
                    "Merge same-segment utility work into one coordinated opening."
                    if match_type == "same_segment"
                    else "Coordinate same-street permits into one trench-sharing window."
                ),
                "reason": (
                    "These permits share the city's GEO_ID (same road segment) and fall within a single coordination window."
                    if match_type == "same_segment"
                    else "These permits are on the same physical road within a single coordination window."
                ),
            })
        return recommendations


    def _temporal_subclusters(group, window_days):
        """Greedy 1-D temporal bucketing.

        Sort by start_date; start a new bucket whenever the next permit's start_date
        is more than `window_days` after the current bucket's end_date.
        Returns list[list[permit]] (each sub-list is one temporal sub-cluster).
        """
        if not group:
            return []
        sorted_group = sorted(group, key=lambda p: p["start_date"])
        buckets = [[sorted_group[0]]]
        for p in sorted_group[1:]:
            current_end = max(q["end_date"] for q in buckets[-1])
            if (p["start_date"] - current_end).days <= window_days:
                buckets[-1].append(p)
            else:
                buckets.append([p])
        return buckets
